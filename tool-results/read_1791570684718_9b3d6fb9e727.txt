   243→   243→        return;
   244→   244→      }
   245→   245→      case "/run":
   246→   246→        if (!arg) {
   247→   247→          appendCommand(name, "usage: /run <command> — try /run neofetch", false);
   248→   248→          return;

   250→   250→        appendCommand(arg, resultToText(await shell.run(arg)));
   251→   251→        return;
   252→   252→      case "/ls":
   253→   253→        appendCommand(name, resultToText(await shell.run("ls")));
   254→   254→        return;
   255→   255→      case "/cat":
   256→   256→        if (!arg) {
   257→   257→          appendCommand(name, "usage: /cat <file> — try /cat README.md", false);

   261→   261→        return;
   262→   262→      case "/npm":

   267→   267→          appendCommand(name, "usage: /git <status|log|push>", false);
   268→   268→          return;
   269→   269→        }

   338→   338→
   339→   339→  /* ---------------- send ---------------- */
   340→   340→
   341→   341→  const send = async (text?: string) => {
   342→   342→    const msg = (text ?? input).trim();

   345→   345→      setInput("");

   349→   349→    }
   350→   350→    setInput("");
   351→   351→    if (taRef.current) taRef.current.style.height = "auto";
   352→   352→    setBusy(true);
   353→   353→
   354→   354→    const userMsg: ChatMessage = { id: `u-${Date.now()}`, role: "user", content: msg };
   355→   355→    const aiId = `a-${Date.now()}`;
   356→   356→    setMessages((m) => [...m, userMsg, { id: aiId, role: "assistant", content: "", model, effort, streaming: true }]);
   357→   357→
   358→   358→    const ctrl = new AbortController();
   359→   359→    abortRef.current = ctrl;
   360→   360→
   361→   361→    let gotDelta = false;
   362→   362→    try {
   363→   363→      const res = await fetch("/api/chat", {
   364→   364→        method: "POST",
   365→   365→        headers: { "Content-Type": "application/json" },
   366→   366→        body: JSON.stringify({ message: msg, model, effort, conversationId: convRef.current }),
   367→   367→        signal: ctrl.signal,
   368→   368→      });
   369→   369→      if (!res.ok || !res.body) {
   370→   370→        const j = await res.json().catch(() => ({}));
   371→   371→        throw new Error(j.error || `HTTP ${res.status}`);
   372→   372→      }
   373→   373→      const reader = res.body.getReader();
   374→   374→      const decoder = new TextDecoder();
   375→   375→      let buf = "";
   376→   376→      for (;;) {
   377→   377→        const { done, value } = await reader.read();
   378→   378→        if (done) break;
   379→   379→        buf += decoder.decode(value, { stream: true });
   380→   380→        const lines = buf.split("\n");
   381→   381→        buf = lines.pop() ?? "";
   382→   382→        for (const line of lines) {
   383→   383→          if (!line.trim()) continue;
   384→   384→          try {
   385→   385→            const evt = JSON.parse(line);
   386→   386→            if (evt.type === "meta" && evt.conversationId) {
   387→   387→              if (!convRef.current) {
   388→   388→                convRef.current = evt.conversationId;
   389→   389→                onConversationCreated(evt.conversationId);
   390→   390→              }
   391→   391→            } else if (evt.type === "delta") {
   392→   392→              gotDelta = true;
   393→   393→              setMessages((m) => m.map((x) => (x.id === aiId ? { ...x, content: x.content + evt.text } : x)));
   394→   394→            } else if (evt.type === "error") {
   395→   395→              throw new Error(evt.error);
   396→   396→            }
   397→   397→          } catch (e) {
   398→   398→            if (e instanceof Error && e.message !== "Unexpected end of JSON input") {
   399→   399→              if (e.message.startsWith("HTTP") || e.message.includes("failed")) throw e;
   400→   400→            }
   401→   401→          }
   402→   402→        }
   403→   403→      }
   404→   404→    } catch (err) {
   405→   405→      const aborted = err instanceof DOMException && err.name === "AbortError";
   406→   406→      if (!aborted && !gotDelta) {
   407→   407→        // static hosting (e.g. GitHub Pages) has no backend — pick the best brain:
   408→   408→        // 1) real Claude via the Anthropic API when the model + key are available
   409→   409→        // 2) built-in demo brain otherwise
   410→   410→        const provider = currentModel?.provider;
   411→   411→        const anthropicKey = useSettings.getState().connections.anthropicKey;
   412→   412→        if (provider === "anthropic" && anthropicKey) {
   413→   413→          const sys =
   414→   414→            `You are ${currentModel?.name ?? "Claude"}, running inside ChatUltra, a dark Codex-style AI studio. ` +
   415→   415→            "Answer in Markdown, always wrap code in fenced blocks with a language tag, and for any web page, game or UI request " +
   416→   416→            "produce a COMPLETE single-file HTML document (inline CSS/JS) inside one ```html block. " +
   417→   417→            `Effort level: ${eff.label}. ${eff.directive}`;
   418→   418→          const turns = [
   419→   419→            ...messages.slice(-8).map((m) => ({ role: m.role, content: m.content })),
   420→   420→            { role: "user" as const, content: msg },
   421→   421→          ];
   422→   422→          const r = await callClaude(anthropicKey, turns, sys, 4096);
   423→   423→          if (r.ok && r.text) {
   424→   424→            await streamDemoReply(
   425→   425→              r.text,
   426→   426→              (text) => setMessages((m) => m.map((x) => (x.id === aiId ? { ...x, content: text } : x))),
