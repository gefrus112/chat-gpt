   108→export function ChatView({ models, model, effort, conversationId, terminalSignal, onModel, onEffort, onConversationCreated, onOpenSettings, onOpenAccount, onNewChat }: ChatViewProps) {

   110→  const [input, setInput] = useState("");

   112→  const [preview, setPreview] = useState<{ code: string; title: string } | null>(null);
   113→  const [termOpen, setTermOpen] = useState(false);
   114→  const [webAccess, setWebAccess] = useState(false);
   115→  const [listening, setListening] = useState(false);
   116→  const [effortOpen, setEffortOpen] = useState(false);
   117→  const [slashSel, setSlashSel] = useState(0);
   118→  const scrollRef = useRef<HTMLDivElement>(null);
   119→  const taRef = useRef<HTMLTextAreaElement>(null);
   120→  const abortRef = useRef<AbortController | null>(null);
   121→  const convRef = useRef<string | null>(conversationId);
   122→  const fileRef = useRef<HTMLInputElement>(null);

   124→
   125→  const shellRef = useRef<ChatShell | null>(null);
   126→  if (!shellRef.current) shellRef.current = new ChatShell();

   128→
   129→  const gh = useSettings((s) => s.gh);

   131→  const appearance = useSettings((s) => s.appearance);
   132→
   133→  useEffect(() => {

   135→  }, [conversationId]);

   137→  useEffect(() => {

   147→  useEffect(() => {

   152→      }
   153→    };
   154→    window.addEventListener("keydown", onWin);
   155→    return () => window.removeEventListener("keydown", onWin);
   156→  }, []);
   157→
   158→  const currentModel = models.find((m) => m.id === model) ?? models[0];
   159→  const eff = findEffort(effort);

   161→  const openPreview = useCallback((code: string, title?: string) => {

   172→    if (!conversationId) {
   173→      setMessages([]);

   175→    }
   176→    let alive = true;
   177→    fetch(`/api/conversations/${conversationId}`)
   178→      .then((r) => r.json())
   179→      .then((d) => {
   180→        if (!alive || !d.conversation) return;
   181→        setMessages(
   182→          d.conversation.messages.map((m: { id: string; role: string; content: string; model?: string; effort?: string }) => ({
   183→            id: m.id,
   184→            role: m.role as "user" | "assistant",
   185→            content: m.content,
   186→            model: m.model,
   187→            effort: m.effort,

   202→  /* ---------------- slash commands ---------------- */

   204→  const appendCommand = (cmd: string, output: string, ok = true) => {
   205→    setMessages((m) => [...m, { id: `c-${Date.now()}`, role: "user", kind: "command", cmd, output, ok, content: "" }]);
   206→  };
   207→
   208→  const handleSlash = async (raw: string) => {
   209→    const text = raw.trim();
   210→    const [name, ...rest] = text.split(/\s+/);
   211→    const arg = rest.join(" ").trim();
   212→
   213→    switch (name) {
   214→      case "/help":
   215→        appendCommand(
   216→          name,
   217→          "ChatUltra commands:\n" + SLASH_COMMANDS.map((c) => `  ${c.name.padEnd(18)} ${c.desc}`).join("\n") +
   218→            "\n\nShell commands (use /run or the terminal): ls cd cat echo touch mkdir rm run build git npm node python ps df uname neofetch whoami date history"
   219→        );
   220→        return;
   221→      case "/models":
   222→        appendCommand(name, "loaded models:\n" + models.map((m) => `  - ${m.name} (${PROVIDER_LABEL[m.provider] ?? m.provider})${m.id === model ? "  [active]" : ""}`).join("\n"));
   223→        return;
   224→      case "/model": {
   225→        const q = arg.toLowerCase();
   226→        const hit = models.find((m) => m.id === arg || m.name.toLowerCase() === q) ?? models.find((m) => m.name.toLowerCase().includes(q) && q.length > 0);
   227→        if (!hit) {
   228→          appendCommand(text, `model not found: "${arg}"\navailable:\n` + models.map((m) => `  - ${m.name}`).join("\n"), false);
   229→          return;
   230→        }
   231→        onModel(hit.id);
   232→        appendCommand(text, `[ok] switched to ${hit.name}`);
   233→        return;
   234→      }
   235→      case "/effort": {
   236→        const e = EFFORTS.find((x) => x.id === arg.toLowerCase());
   237→        if (!e) {
   238→          appendCommand(text, `unknown effort "${arg}" — levels: ${EFFORTS.map((x) => x.id).join(", ")}`, false);
   239→          return;
   240→        }
   241→        onEffort(e.id);
   242→        appendCommand(text, `[ok] reasoning effort set to ${e.label}`);
   243→        return;
   244→      }
   245→      case "/run":
   246→        if (!arg) {
   247→          appendCommand(name, "usage: /run <command> — try /run neofetch", false);
   248→          return;
   249→        }
   250→        appendCommand(arg, resultToText(await shell.run(arg)));
   251→        return;
   252→      case "/ls":
   253→        appendCommand(name, resultToText(await shell.run("ls")));
   254→        return;
   255→      case "/cat":
   256→        if (!arg) {
   257→          appendCommand(name, "usage: /cat <file> — try /cat README.md", false);
   258→          return;
   259→        }
   260→        appendCommand(`${name} ${arg}`, resultToText(await shell.run(`cat ${arg}`)));
   261→        return;
   262→      case "/npm":
   263→        appendCommand(text, resultToText(await shell.run(`npm ${arg || "--help"}`.trim())));
   264→        return;
   265→      case "/git":
   266→        if (!arg) {
   267→          appendCommand(name, "usage: /git <status|log|push>", false);
   268→          return;
   269→        }
   270→        appendCommand(text, resultToText(await shell.run(`git ${arg}`)));
   271→        return;
   272→      case "/terminal":
   273→        setTermOpen(true);
   274→        appendCommand(name, "[ok] terminal opened — type `help` inside");
   275→        return;
   276→      case "/web":
   277→        setWebAccess((w) => {
   278→          appendCommand(name, `[ok] web access ${!w ? "enabled" : "disabled"}`);
   279→          return !w;
   280→        });
   281→        return;
   282→      case "/game":
   283→        appendCommand(name, arg ? `[ok] building game: ${arg}` : "usage: /game <description> — try /game neon pong", Boolean(arg));
   284→        if (arg) send(`Build a complete playable single-file HTML game: ${arg}. Full HTML in one code block.`);
   285→        return;
   286→      case "/export": {
   287→        const md =
   288→          "# ChatUltra chat export\n\n" +
   289→          messages
   290→            .map((m) =>
   291→              m.kind === "command"
   292→                ? `\`\`\`console\n$ ${m.cmd}\n${m.output ?? ""}\n\`\`\``
   293→                : `**${m.role === "user" ? "You" : m.model || "ChatUltra"}**\n\n${m.content}`
   294→            )
   295→            .join("\n\n---\n\n");
   296→        try {
   297→          const blob = new Blob([md], { type: "text/markdown" });
   298→          const a = document.createElement("a");
   299→          a.href = URL.createObjectURL(blob);
   300→          a.download = "chatultra-chat.md";
   301→          a.click();
   302→          URL.revokeObjectURL(a.href);
   303→          appendCommand(name, "[ok] exported chatultra-chat.md");
   304→        } catch {
   305→          appendCommand(name, "[err] export failed in this browser", false);
   306→        }
   307→        return;
   308→      }
   309→      case "/push":
   310→        onOpenSettings("github");
   311→        appendCommand(name, "[ok] opening GitHub settings — token + repo are prefilled");
   312→        return;
   313→      case "/account":
   314→        onOpenAccount();
   315→        appendCommand(name, "[ok] opening account profile");
   316→        return;
   317→      case "/settings":
   318→        onOpenSettings("appearance");
   319→        appendCommand(name, "[ok] opening settings");
   320→        return;
   321→      case "/theme": {
   322→        const a = ACCENTS.find((x) => x.id === arg.toLowerCase());
   323→        if (!a) {
   324→          appendCommand(text, `unknown accent "${arg}" — options: ${ACCENTS.map((x) => x.id).join(", ")}`, false);
   325→          return;
   326→        }
   327→        setAppearance({ accent: a.hex });
   328→        appendCommand(text, `[ok] accent set to ${a.label} (${a.hex})`);
   329→        return;
   330→      }
   331→      case "/clear":
   332→        onNewChat();
   333→        return;
   334→      default:
   335→        appendCommand(name, `unknown command: ${name} — type /help`, false);
   336→    }
   337→  };
   338→
   339→  /* ---------------- send ---------------- */
   340→
   341→  const send = async (text?: string) => {
   342→    const msg = (text ?? input).trim();
   343→    if (!msg || busy) return;
   344→    if (msg.startsWith("/")) {
   345→      setInput("");
   346→      if (taRef.current) taRef.current.style.height = "auto";
   347→      await handleSlash(msg);
   348→      return;
   349→    }
   350→    setInput("");
   351→    if (taRef.current) taRef.current.style.height = "auto";
   352→    setBusy(true);
   353→
   354→    const userMsg: ChatMessage = { id: `u-${Date.now()}`, role: "user", content: msg };
   355→    const aiId = `a-${Date.now()}`;
   356→    setMessages((m) => [...m, userMsg, { id: aiId, role: "assistant", content: "", model, effort, streaming: true }]);
   357→
   358→    const ctrl = new AbortController();
   359→    abortRef.current = ctrl;
   360→
   361→    let gotDelta = false;
   362→    try {
   363→      const res = await fetch("/api/chat", {
   364→        method: "POST",
   365→        headers: { "Content-Type": "application/json" },
   366→        body: JSON.stringify({ message: msg, model, effort, conversationId: convRef.current }),
   367→        signal: ctrl.signal,
   368→      });
   369→      if (!res.ok || !res.body) {
   370→        const j = await res.json().catch(() => ({}));
   371→        throw new Error(j.error || `HTTP ${res.status}`);
   372→      }
   373→      const reader = res.body.getReader();
   374→      const decoder = new TextDecoder();
   375→      let buf = "";
   376→      for (;;) {
   377→        const { done, value } = await reader.read();
   378→        if (done) break;
   379→        buf += decoder.decode(value, { stream: true });
   380→        const lines = buf.split("\n");
   381→        buf = lines.pop() ?? "";
   382→        for (const line of lines) {
   383→          if (!line.trim()) continue;
   384→          try {
   385→            const evt = JSON.parse(line);
   386→            if (evt.type === "meta" && evt.conversationId) {
   387→              if (!convRef.current) {
   388→                convRef.current = evt.conversationId;
   389→                onConversationCreated(evt.conversationId);
   390→              }
   391→            } else if (evt.type === "delta") {
   392→              gotDelta = true;
   393→              setMessages((m) => m.map((x) => (x.id === aiId ? { ...x, content: x.content + evt.text } : x)));
   394→            } else if (evt.type === "error") {
   395→              throw new Error(evt.error);
   396→            }
   397→          } catch (e) {
   398→            if (e instanceof Error && e.message !== "Unexpected end of JSON input") {
   399→              if (e.message.startsWith("HTTP") || e.message.includes("failed")) throw e;
   400→            }
   401→          }
   402→        }
   403→      }
   404→    } catch (err) {
   405→      const aborted = err instanceof DOMException && err.name === "AbortError";
   406→      if (!aborted && !gotDelta) {
   407→        // static hosting (e.g. GitHub Pages) has no backend — pick the best brain:
   408→        // 1) real Claude via the Anthropic API when the model + key are available
   409→        // 2) built-in demo brain otherwise
   410→        const provider = currentModel?.provider;
   411→        const anthropicKey = useSettings.getState().connections.anthropicKey;
   412→        if (provider === "anthropic" && anthropicKey) {
   413→          const sys =
   414→            `You are ${currentModel?.name ?? "Claude"}, running inside ChatUltra, a dark Codex-style AI studio. ` +
   415→            "Answer in Markdown, always wrap code in fenced blocks with a language tag, and for any web page, game or UI request " +
   416→            "produce a COMPLETE single-file HTML document (inline CSS/JS) inside one ```html block. " +
   417→            `Effort level: ${eff.label}. ${eff.directive}`;
   418→          const turns = [
   419→            ...messages.slice(-8).map((m) => ({ role: m.role, content: m.content })),
   420→            { role: "user" as const, content: msg },
   421→          ];
   422→          const r = await callClaude(anthropicKey, turns, sys, 4096);
   423→          if (r.ok && r.text) {
   424→            await streamDemoReply(
   425→              r.text,
   426→              (text) => setMessages((m) => m.map((x) => (x.id === aiId ? { ...x, content: text } : x))),
   427→              ctrl.signal
   428→            );
   429→          } else {
   430→            toast({ title: "Claude call failed — switched to the local engine", description: r.error });
   431→            const reply = demoReply(msg);
   432→            await streamDemoReply(
   433→              reply,
   434→              (text) => setMessages((m) => m.map((x) => (x.id === aiId ? { ...x, content: text } : x))),
   435→              ctrl.signal
   436→            );
   437→          }
   438→        } else {
   439→          if (provider === "anthropic" && !anthropicKey) {
   440→            toast({
   441→              title: "Demo engine",
   442→              description: `Paste an Anthropic API key in Settings > Connections to chat with ${currentModel?.name} for real.`,
   443→            });
   444→          }
   445→          const reply = demoReply(msg);
   446→          await streamDemoReply(
   447→            reply,
   448→            (text) => setMessages((m) => m.map((x) => (x.id === aiId ? { ...x, content: text } : x))),
   449→            ctrl.signal
   450→          );
   451→        }
   452→      } else {
   453→        setMessages((m) =>
   454→          m.map((x) =>
   455→            x.id === aiId
   456→              ? { ...x, content: x.content || (aborted ? "_stopped._" : `Error: ${err instanceof Error ? err.message : "Something went wrong — try again."}`) }
   457→              : x
   458→          )
   459→        );
   460→      }
   461→    } finally {
   462→      setMessages((m) => m.map((x) => (x.id === aiId ? { ...x, streaming: false } : x)));
   463→      setBusy(false);
   464→      abortRef.current = null;
   465→    }
   466→  };
   467→
   468→  const stop = () => abortRef.current?.abort();
   469→
   470→  /* ---------------- voice input ---------------- */
   471→
   472→  const startVoice = () => {
   473→    type Rec = { start: () => void; stop: () => void; onresult: ((e: { results: { 0: { 0: { transcript: string } } } }) => void) | null; onend: (() => void) | null; onerror: (() => void) | null; continuous: boolean; interimResults: boolean; lang: string };
   474→    const w = window as unknown as { SpeechRecognition?: new () => Rec; webkitSpeechRecognition?: new () => Rec };
   475→    const Ctor = w.SpeechRecognition ?? w.webkitSpeechRecognition;
   476→    if (!Ctor) {
   477→      toast({ title: "Voice input unavailable", description: "This browser does not support speech recognition — try Chrome or Edge." });
   478→      return;
   479→    }
   480→    if (listening) {
   481→      recRef.current?.stop();
   482→      return;
   483→    }
   484→    const rec = new Ctor();
   485→    rec.continuous = false;
   486→    rec.interimResults = false;
   487→    rec.lang = "en-US";
   488→    rec.onresult = (e) => {
   489→      const t = e.results?.[0]?.[0]?.transcript ?? "";
   490→      if (t) setInput((prev) => (prev ? prev + " " : "") + t);
   491→    };
   492→    rec.onend = () => {
   493→      setListening(false);
   494→      recRef.current = null;
   495→    };
   496→    rec.onerror = () => {
   497→      setListening(false);
   498→      recRef.current = null;
   499→    };
   500→    recRef.current = rec;
   501→    setListening(true);
   502→    rec.start();
   503→  };
   504→
   505→  /* ---------------- attach ---------------- */
   506→
   507→  const onFile = async (e: React.ChangeEvent<HTMLInputElement>) => {
   508→    const f = e.target.files?.[0];
   509→    e.target.value = "";
   510→    if (!f) return;
   511→    if (f.size > 80 * 1024) {
   512→      toast({ title: "File too large", description: "Attach text files under 80 kB." });
   513→      return;
   514→    }
   515→    const text = await f.text();

   593→                  src={asset("/logo.png")}

   595→                  width={110}
   596→                  height={110}
   597→                  className="rounded-3xl border border-white/10 shadow-2xl shadow-cyan-500/20"
   598→                  priority
   599→                  unoptimized
   600→                />
   601→              </div>
   602→              <h1 className="bg-gradient-to-r from-cyan-200 via-white to-violet-300 bg-clip-text text-4xl font-bold tracking-tight text-transparent">
   603→                ChatUltra
   604→              </h1>
   605→              <p className="mt-2 text-[13.5px] text-zinc-400">
   606→                Chat · Build games · Run commands · Ship to GitHub — your Codex-grade AI workspace

   611→                    key={s.title}
   612→                    onClick={() => send(s.prompt)}
   613→                    className="group rounded-xl border border-white/10 bg-white/[0.03] p-3.5 text-left transition hover:border-cyan-400/30 hover:bg-cyan-400/[0.06]"

   617→                      {s.title}
   618→                    </div>
   619→                    <div className="mt-1 line-clamp-2 text-[11.5px] leading-relaxed text-zinc-500 group-hover:text-zinc-400">
   620→                      {s.prompt}
   621→                    </div>
   622→                  </button>
   623→                ))}
   624→              </div>

   626→                <TerminalSquare className="h-3.5 w-3.5" />

   629→            </div>
   630→          ) : (
   631→            /* ---------- messages ---------- */
   632→            <div ref={scrollRef} className="chatultra-scroll min-h-0 flex-1 overflow-y-auto px-4 py-6">

   634→                {messages.map((m) =>
   635→                  m.kind === "command" ? (

   664→                      <div className={cn("min-w-0", m.role === "user" ? "max-w-[85%]" : "max-w-full flex-1")}>
   665→                        {m.role === "user" ? (
   666→                          <div className="rounded-2xl rounded-tr-md border border-cyan-400/20 bg-cyan-400/[0.08] px-4 py-2.5 text-[14px] leading-relaxed text-zinc-100 whitespace-pre-wrap">
   667→                            {m.content}
   668→                          </div>
   669→                        ) : (
   670→                          <div className="rounded-2xl rounded-tl-md border border-white/[0.07] bg-white/[0.02] px-4 py-2">
   671→                            {m.content ? (
   672→                              <Markdown content={m.content} className={fontCls} />
   673→                            ) : (
   674→                              <div className="flex items-center gap-2 py-1 text-[13px] text-zinc-400">
   675→                                <Loader2 className="h-3.5 w-3.5 animate-spin text-cyan-300" />

   679→                            {!m.streaming && m.content && (
   680→                              <div className="mt-1.5 flex items-center gap-2 border-t border-white/5 pt-1.5 text-[10.5px] text-zinc-600">

   690→                          You

   705→                {slashOpen && (
   706→                  <div className="absolute bottom-full left-0 right-0 z-20 mb-2 overflow-hidden rounded-xl border border-white/10 bg-[#0c101c] shadow-2xl shadow-black/60">
   707→                    <div className="chatultra-scroll max-h-[264px] overflow-y-auto p-1.5">
   708→                      {slashFiltered.map((c, i) => (
   709→                        <button
   710→                          key={c.name}
   711→                          onMouseEnter={() => setSlashSel(i)}
   712→                          onClick={() => execFromMenu(c.name)}
   713→                          className={cn(
   714→                            "flex w-full items-center gap-2.5 rounded-lg px-2.5 py-2 text-left transition",
   715→                            i === slashSel ? "bg-cyan-400/10" : "hover:bg-white/[0.05]"

   717→                        >
   718→                          <span className="w-32 shrink-0 font-mono text-[12.5px] font-medium text-cyan-300">{c.name}</span>
   719→                          <span className="truncate text-[12px] text-zinc-400">{c.desc}</span>
   720→                        </button>

   722→                    </div>
   723→                    <div className="border-t border-white/[0.07] px-3 py-1.5 text-[10px] text-zinc-600">Tab to complete · Enter to run · Esc to dismiss</div>

   727→                <textarea
   728→                  ref={taRef}
   729→                  value={input}

   731→                  onKeyDown={onTaKey}
   732→                  rows={1}
   733→                  placeholder="Ask ChatUltra to build, explain or ship anything — or type / for commands"

   744→                    models={models}

   750→                    onEditCustom={() => onOpenSettings("models")}

   798→                    onClick={() => setTermOpen((t) => !t)}
   799→                    className={cn(termOpen ? toolBtnActive : toolBtn)}
   800→                    title="Terminal — run commands (Ctrl+`)"
   801→                    aria-pressed={termOpen}
   802→                  >
   803→                    <SquareTerminal className="h-4 w-4" />

   806→                  {/* attach file */}
